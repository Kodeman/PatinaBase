The surcharge arithmetic, as this page computes it 
// packages/shared/src/invoice/index.ts:169-223 — integer half-up
const fee = (cents, bps) => Math.floor((cents * bps + 5000) / 10000);

function surchargeCents(balanceCents, method, cardBps) {
  if (method === 'us_bank_account') return Math.min(fee(balanceCents, 80), 500);
  if (method === 'card')            return fee(balanceCents, cardBps);
  return 0; // check
}
