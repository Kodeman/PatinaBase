    // MARK: - Media, as the device writes it

    static func image(_ index: Int) -> UIImage {
        let size = CGSize(width: 96, height: 128)
        let format = UIGraphicsImageRendererFormat()
        format.scale = 1
        return UIGraphicsImageRenderer(size: size, format: format).image { context in
            UIColor(hue: CGFloat(index % 12) / 12, saturation: 0.45, brightness: 0.8, alpha: 1).setFill()
            context.fill(CGRect(origin: .zero, size: size))
            UIColor.white.setFill()
            context.fill(CGRect(x: 8 + (index * 7) % 56, y: 24, width: 28, height: 48))
        }
    }

    static func heic(_ index: Int) throws -> Data {
        let data = NSMutableData()
        guard let cgImage = image(index).cgImage,
              let destination = CGImageDestinationCreateWithData(
                data, UTType.heic.identifier as CFString, 1, nil) else {
            throw SeedFailure(description: "no HEIC destination")
        }
        CGImageDestinationAddImage(destination, cgImage,
                                   [kCGImageDestinationLossyCompressionQuality: 0.6] as CFDictionary)
        guard CGImageDestinationFinalize(destination) else {
            throw SeedFailure(description: "HEIC encode failed")
        }
        return data as Data
    }

    /// A silent AAC segment, written where SpeechVoiceNoteService writes one:
    /// straight into the media directory under VoiceRecordingPolicy's name.
    static func writeSilentSegment(to url: URL, seconds: Double) throws {
        let settings: [String: Any] = [
            AVFormatIDKey: kAudioFormatMPEG4AAC,
            AVSampleRateKey: 16_000,
            AVNumberOfChannelsKey: 1
        ]
        let file = try AVAudioFile(forWriting: url, settings: settings,
                                   commonFormat: .pcmFormatFloat32, interleaved: false)
        let frames = AVAudioFrameCount(seconds * 16_000)
        guard let buffer = AVAudioPCMBuffer(pcmFormat: file.processingFormat,
                                            frameCapacity: frames) else {
            throw SeedFailure(description: "no PCM buffer")
        }
        buffer.frameLength = frames
        try file.write(from: buffer)
    }

    /// The transfer transitions LocalCaptureSyncService makes, in its order.
    static func enqueue(_ s: Specimen) {
        s.status = .ready
        s.applyTransferState(CaptureTransferState(phase: .queued, retryCount: s.retryCount))
    }

    static func upload(_ s: Specimen, progress: Int) {
        s.applyTransferState(CaptureTransferState(phase: .uploading, progress: progress,
                                                  retryCount: s.retryCount))
    }

    static func awaitConfirmation(_ s: Specimen) {
        s.applyTransferState(CaptureTransferState(phase: .awaitingConfirmation, progress: 100,
                                                  retryCount: s.retryCount))
    }

    static func commit(_ s: Specimen) {
        enqueue(s)
        upload(s, progress: 100)
        awaitConfirmation(s)
        s.applyTransferState(CaptureTransferState(
            phase: .complete, progress: 100, retryCount: s.retryCount,
            receiptID: UUID().uuidString.lowercased()))
    }

    /// Committed captures whose wave-4 write lanes keep them in `outbox()`:
    /// every FieldWriteState on the margin, punch and degrade lanes, and every
    /// ProjectPlacementState, spread over five rows, plus one fully settled row.
    func seedCommittedLanes(_ store: CaptureStore, owner: CaptureOwnerIdentity) throws {
        var rows: [Specimen] = []
        for index in 0..<6 {
            let row = try capture(store, owner: owner, title: "Committed capture \(index + 1)",
                                  photos: 1, category: .decor,
                                  measurements: [(.width, 300 + Double(index), .arkit)])
            Self.commit(row)
            row.configureProjectPlacement(projectID: Self.mapleStreet, roomID: Self.livingRoom,
                                          slotID: UUID().uuidString.lowercased(), category: "decor")
            row.requestMarginNote(noteID: UUID())
            row.requestPunchTask(taskID: UUID(), owner: "designer", partyID: nil)
            row.requestDegradeNote(noteID: UUID(), body: "Couldn't hand this to the GC — kept as a note.")
            rows.append(row)
        }
        let placed = { (row: Specimen) in
            row.markProjectPlacementStarted()
            row.applyProjectPlacementReceipt(ProjectPlacementReceipt(
                projectID: UUID(uuidString: Self.mapleStreet)!, ffeItemID: UUID(), specID: UUID(),
                productID: UUID(), roomID: UUID(uuidString: Self.livingRoom), placement: "created"))
        }

        // 1 — every lane pending
        label(rows[0], "Specimen · committed; placement pending, margin/punch/degrade pending")

        // 2 — every lane in flight
        rows[1].markProjectPlacementStarted()
        rows[1].markMarginNoteStarted()
        rows[1].markPunchTaskStarted()
        rows[1].markDegradeNoteStarted()
        label(rows[1], "Specimen · committed; placement placing, margin/punch/degrade writing")

        // 3 — every lane failed once
        rows[2].markProjectPlacementStarted()
        rows[2].markProjectPlacementFailed("The request timed out.")
        rows[2].markMarginNoteFailed("The request timed out.")
        rows[2].markPunchTaskFailed("The request timed out.")
        rows[2].markDegradeNoteFailed("The request timed out.")
        label(rows[2], "Specimen · committed; placement failed, margin/punch/degrade failed (retry 1)")

        // 4 — refused, and re-placed after committing (replay pending)
        placed(rows[3])
        rows[3].markMarginNoteRefused("new row violates row-level security policy")
        rows[3].markPunchTaskRefused("new row violates row-level security policy")
        rows[3].markDegradeNoteRefused("new row violates row-level security policy")
        rows[3].place(projectID: Self.highPoint, projectRoomID: nil, room: "Showroom 214")
        label(rows[3], "Specimen · committed; placed, margin/punch/degrade refused, placementReplayPending")

        // 5 — unwritable: margin through the retry ceiling, the others first-attempt
        placed(rows[4])
        for _ in 0..<FieldWriteGate.retryCeiling {
            rows[4].markMarginNoteFailed("Internal Server Error")
        }
        rows[4].markPunchTaskUnwritable("null value in column \"project_id\" violates not-null constraint")
        rows[4].markDegradeNoteUnwritable("insert or update on table violates foreign key constraint")
        label(rows[4], "Specimen · committed; placed, margin unwritable (retry ceiling), punch/degrade unwritable")

        // 6 — fully settled; not in outbox()
        placed(rows[5])
        rows[5].markMarginNoteStarted(); rows[5].markMarginNoteWritten()
        rows[5].markPunchTaskStarted(); rows[5].markPunchTaskWritten()
        rows[5].markDegradeNoteStarted(); rows[5].markDegradeNoteWritten()
        label(rows[5], "Specimen · committed and fully synced; placed, every lane written")
        try store.save()
    }

    // MARK: - Site-scan uploads (ScanUploadRecord)

    static func artifacts(_ statuses: [ScanArtifactUploadState.Status]) -> [ScanArtifactUploadState] {
        let kinds: [(String, String, String, String?)] = [
            ("bundleManifest", "manifest.json", "application/json", "bundle_manifest_url"),
            ("usdz", "room.usdz", "model/vnd.usdz+zip", "usdz_url"),
            ("capturedRoomJson", "captured-room.json", "application/json", "captured_room_json_url"),
            ("heroFrame", "hero.heic", "image/heic", nil),
            ("coverageHeatmap", "coverage.png", "image/png", nil)
        ]
        return zip(kinds, statuses).map { kind, status in
            let uploaded = status == .uploaded
            return ScanArtifactUploadState(
                kind: kind.0, relativePath: kind.1, mimeType: kind.2,
                storagePath: uploaded ? "room-scans/\(ownerA.userID)/\(kind.1)" : nil,
                remoteUrl: nil, sha256: uploaded ? String(repeating: "a", count: 64) : nil,
                column: kind.3, status: status,
                attempts: status == .failed ? 2 : (status == .pending ? 0 : 1),
                lastError: status == .failed ? "The network connection was lost." : nil,
                shadowUploaded: uploaded ? true : nil,
                shadowSha256: uploaded ? String(repeating: "a", count: 64) : nil,
                shadowMatched: uploaded ? true : nil)
        }
    }

    func seedScanUploads(_ store: CaptureStore) throws {
        let cases: [(String, [ScanArtifactUploadState.Status], CaptureTransferState)] = [
            ("queued", [.pending, .pending, .pending, .pending, .pending],
             CaptureTransferState(phase: .queued)),
            ("uploading", [.uploaded, .uploading, .pending, .pending, .skipped],
             CaptureTransferState(phase: .uploading, progress: 20)),
            ("awaitingConfirmation", [.uploaded, .uploaded, .uploaded, .uploaded, .skipped],
             CaptureTransferState(phase: .awaitingConfirmation, progress: 100)),
            ("retryableFailure", [.uploaded, .failed, .pending, .pending, .pending],
             CaptureTransferState(phase: .retryableFailure, progress: 20,
                                  errorMessage: "The network connection was lost.", retryCount: 1)),
            ("rejected", [.uploaded, .failed, .failed, .pending, .pending],
             CaptureTransferState(phase: .rejected, progress: 20,
                                  errorMessage: "This scan is missing its room model.")),
            ("complete", [.uploaded, .uploaded, .uploaded, .uploaded, .uploaded],
             CaptureTransferState(phase: .complete, progress: 100,
                                  receiptID: UUID().uuidString.lowercased()))
        ]
        for (name, statuses, transfer) in cases {
            let key = SiteScanBundleHome.relativeKey(
                for: URL(fileURLWithPath: "/SiteScans/site-scan-\(UUID().uuidString.lowercased())"))
            let record = try store.insertScanUploadRecord(ScanUploadRecord(
                bundlePath: key, scanID: UUID().uuidString, roomID: Self.livingRoom,
                name: "Living — \(name)", projectID: Self.mapleStreet,
                projectRoomID: Self.livingRoom, owner: Self.ownerA))
            store.updateScanUploadRecord(record, artifacts: Self.artifacts(statuses),
                                         transfer: transfer)
            labels[key] = "ScanUploadRecord · \(name)"
        }
    }

    // MARK: - Site-request guest deliveries (SiteRequestOutboxRecord)

    func seedSiteRequests(_ store: CaptureStore) throws {
        let plan: [(String, [SiteRequestOutboxState], SiteRequestOutboxTerminalReason?)] = [
            ("queued", [], nil),
            ("uploading", [.uploading], nil),
            ("awaitingReceipt", [.uploading, .awaitingReceipt], nil),
            ("delivered", [.uploading, .awaitingReceipt, .delivered], nil),
            ("failed", [.uploading, .failed], nil),
            ("terminal", [.uploading, .terminal], .requestChanged)
        ]
        for (name, path, reason) in plan {
            let deliveryID = UUID()
            // SiteRequestScreens.captureProof + enqueue(submission:mediaPaths:)
            let proof = try store.writeMedia(try Self.heic(50 + labels.count),
                                             filename: "site-proof-\(UUID().uuidString).heic")
            let submission = SiteDeliverySubmission(
                requestID: UUID().uuidString.lowercased(), itemID: UUID().uuidString.lowercased(),
                itemVersionID: UUID().uuidString.lowercased(), clientDeliveryID: deliveryID,
                uploadIDs: [UUID().uuidString.lowercased()],
                skippedShotLabels: name == "failed" ? ["Behind the door"] : [])
            let encoder = JSONEncoder()
            encoder.dateEncodingStrategy = .iso8601
            let payload = try encoder.encode(submission)
            let payloadURL = try store.writeMedia(
                payload, filename: "site-delivery-\(deliveryID.uuidString).json")
            let record = SiteRequestOutboxRecord(
                clientDeliveryID: deliveryID, requestID: submission.requestID,
                itemID: submission.itemID, itemVersionID: submission.itemVersionID,
                payloadPath: payloadURL.path, mediaPaths: [proof.path],
                checksumSHA256: SiteRequestChecksum.sha256(payload))
            try store.enqueueSiteRequestDelivery(record)
            for state in path {
                try store.transitionSiteRequestDelivery(
                    record, to: state,
                    error: state == .failed ? "The request timed out." : nil,
                    serverDeliverableID: state == .delivered ? UUID().uuidString.lowercased() : nil,
                    terminalReason: state == .terminal ? reason : nil)
            }
            labels[deliveryID.uuidString] = "SiteRequestOutboxRecord · \(name)"
        }
    }

    func seedVisitCloses(_ store: CaptureStore) throws {
        let ownerUserID = try #require(UUID(uuidString: Self.ownerA.userID))
        for (index, entry) in Self.hoursOutcomes.enumerated() {
            let visitID = UUID()
            let end = Date().addingTimeInterval(-Double(index) * 3_600)
            // V4VisitReviewScreen.logTheHours
            let record = FieldVisitCloseRecord(
                visitID: visitID, timeEntryID: UUID(), projectID: Self.mapleStreet,
                ownerUserID: ownerUserID.uuidString, startedAt: end.addingTimeInterval(-95 * 60),
                endedAt: end, durationMinutes: 95, billable: index % 3 != 2)
            store.context.insert(record)
            try store.save()
            // VisitCloseOutboxDrainer.drain
            if entry.0 == "writing" { record.state = .writing }
            for outcome in entry.1 {
                record.state = .writing
                VisitCloseOrchestrator.apply(outcome, to: record, now: Date())
            }
            try store.save()
            labels[visitID.uuidString] = "FieldVisitCloseRecord · \(entry.0)"
                + (record.billable ? "" : " (not billable)")
        }
    }

                notes: index == 0 ? "Maple St → High Point" : "",
                rateRole: index == 1 ? .leadDesigner : nil)
            let entryID = UUID()

            store.context.insert(record)

    // MARK: - Manifest

    func writeManifest(_ store: CaptureStore) throws {
        let storeURL = try #require(store.container.configurations.first?.url)
        let mediaDirectory = store.mediaDirectory()
        var media: [String: String] = [:]
        for name in try FileManager.default.contentsOfDirectory(atPath: mediaDirectory.path) {
            let data = try Data(contentsOf: mediaDirectory.appendingPathComponent(name))
            #expect(!data.isEmpty, "media file \(name) is empty")
            media[name] = StoreFixtureProjection.sha256(data)
        }

        let userInitiated = VisitCloseDrainTrigger.userInitiated
        let queries: [String: [String]] = [
            "outbox()": store.outbox().map(\.id.uuidString).sorted(),
            "outbox(owner: A)": store.outbox(owner: Self.ownerA).map(\.id.uuidString).sorted(),
            "outbox(owner: B)": store.outbox(owner: Self.ownerB).map(\.id.uuidString).sorted(),
            "visitCloseOutbox(owner: A)":
                store.visitCloseOutbox(owner: Self.ownerA).map(\.visitID.uuidString).sorted(),
            "visitCloseDrainable(owner: A, userInitiated)":
                VisitCloseOrchestrator.drainable(store.visitCloseOutbox(owner: Self.ownerA),
                                                 at: Date(), trigger: userInitiated)
                    .map(\.visitID.uuidString).sorted(),
            "timeEntryOutbox(owner: A)":
                store.timeEntryOutbox(owner: Self.ownerA).map(\.entryID.uuidString).sorted(),
            "timeEntryDrainable(owner: A, userInitiated)":
                TimeEntryOutboxOrchestrator.drainable(store.timeEntryOutbox(owner: Self.ownerA),
                                                      at: Date(), trigger: userInitiated)
                    .map(\.entryID.uuidString).sorted(),
            "siteRequestOutbox()": store.siteRequestOutbox().map(\.clientDeliveryID.uuidString).sorted(),
            "scanUploadRecords(owner: A)":
                store.scanUploadRecords(owner: Self.ownerA).map(\.bundlePath).sorted(),
            "scanUploadRecords(owner: A, includeComplete: true)":
                store.scanUploadRecords(owner: Self.ownerA, includeComplete: true)
                    .map(\.bundlePath).sorted(),
            "scanBundlePathsProtectedFromSweep()": store.scanBundlePathsProtectedFromSweep().sorted()
        ]

        let manifest = StoreFixtureManifest(
            sourceCommit: "b1447ba7da595c94e724611c28b4e8439267f0dc",
            build: "Patina Field 0.1 (6)",
            storeRung: store.openReport.persistence.rawValue,
            storeFilename: storeURL.lastPathComponent,
            seededStorePath: storeURL.path,
            seededMediaPath: mediaDirectory.path,
            rows: try StoreFixtureProjection.snapshot(store.context),
            media: media,
            labels: labels,
            queries: queries)
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys, .withoutEscapingSlashes]
        let out = storeURL.deletingLastPathComponent().appendingPathComponent("fixture-manifest.json")
        try encoder.encode(manifest).write(to: out, options: .atomic)
        print("SQ195 store=\(storeURL.path) media=\(mediaDirectory.path) manifest=\(out.path)")
    }
